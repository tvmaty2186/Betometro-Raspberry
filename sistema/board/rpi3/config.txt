# Betómetro — firmware de la Raspberry Pi 3 B+ (64 bits)
start_file=start.elf
fixup_file=fixup.dat
kernel=Image
arm_64bit=1

# Arranque rápido y sin el arcoíris del firmware
disable_splash=1
boot_delay=0

# Pantalla: placa de video con KMS y la Touch Display 2 de 7" (720×1280, de pie)
dtoverlay=vc4-kms-v3d
dtoverlay=vc4-kms-dsi-ili9881-7inch
gpu_mem=128

# GPS por el UART bueno (ttyAMA0): se apaga el Bluetooth
enable_uart=1
dtoverlay=disable-bt

# Sonido por la salida de auriculares, con menos ruido
dtparam=audio=on
audio_pwm_mode=2

# Vigilante (watchdog) del procesador
dtparam=watchdog=on
